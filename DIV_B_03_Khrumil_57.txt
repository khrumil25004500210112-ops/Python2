"""Program 1.21: Armstrong number checker and finder."""
def is_armstrong(number):
    digits = str(number)
    return number == sum(int(digit) ** len(digits) for digit in digits)

number = int(input("Enter a number: "))
if is_armstrong(number):
    print(number, "is an Armstrong number.")
else:
    print(number, "is not an Armstrong number.")
print("Armstrong numbers from 1 to 999:")
print(*[n for n in range(1, 1000) if is_armstrong(n)])
'''
Enter a number: 12
12 is not an Armstrong number.
Armstrong numbers from 1 to 999:
1 2 3 4 5 6 7 8 9 153 370 371 407

"""Program 3.2: stack using a list."""
stack = []
def push(item): stack.append(item)
def pop(): return stack.pop() if stack else "Stack is empty"
def peek(): return stack[-1] if stack else "Stack is empty"
def is_empty(): return not stack
def display(): print("Stack:", stack)

for item in [10, 20, 30, 40, 50]: push(item)
display()
print("Peek:", peek())
print("Popped:", pop(), pop())
display()
print("Is empty:", is_empty())
'''
Stack: [10, 20, 30, 40, 50]
Peek: 50
Popped: 50 40
Stack: [10, 20, 30]
Is empty: False
'''
